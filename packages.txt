# Manual checklist of system packages to install on a new machine.
# The desktop section uses Ubuntu 26.04 apt names; other sections still use Arch names.
# CLI tools come from mise (~/.config/mise/config.toml), installed on chezmoi apply.

playerctl   # Media player controller

sesh-bin  # Session manager
